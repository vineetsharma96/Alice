cmake_minimum_required(VERSION 3.20)
project(UnderworldLab VERSION 0.8.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/core/world.cpp" WORLD_HASH)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/core/world.hpp" HEADER_HASH)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/core/bindings.cpp" BINDINGS_HASH)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/core/bodies.cpp" BODIES_HASH)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/core/teaching.cpp" TEACHING_HASH)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/core/thermal.hpp" THERMAL_HEADER_HASH)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/core/thermal.cpp" THERMAL_HASH)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/core/thermal_world.cpp" THERMAL_WORLD_HASH)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/core/culture.hpp" CULTURE_HEADER_HASH)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/core/culture.cpp" CULTURE_HASH)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/core/supervision.cpp" SUPERVISION_HASH)
string(SHA256 BUILD_HASH "${WORLD_HASH}${HEADER_HASH}${BINDINGS_HASH}${BODIES_HASH}${TEACHING_HASH}${THERMAL_HEADER_HASH}${THERMAL_HASH}${THERMAL_WORLD_HASH}${CULTURE_HEADER_HASH}${CULTURE_HASH}${SUPERVISION_HASH}${CMAKE_CXX_COMPILER_ID}${CMAKE_CXX_COMPILER_VERSION}${CMAKE_BUILD_TYPE}${CMAKE_CXX_FLAGS}${CMAKE_SYSTEM_NAME}${CMAKE_SYSTEM_PROCESSOR}")
add_library(underworld_sim STATIC core/world.cpp core/bodies.cpp core/teaching.cpp core/thermal.cpp core/thermal_world.cpp core/culture.cpp core/supervision.cpp)
set_target_properties(underworld_sim PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_definitions(underworld_sim PUBLIC UW_BUILD_ID="${BUILD_HASH}")
target_include_directories(underworld_sim PUBLIC core)
if(MSVC)
    target_compile_options(underworld_sim PRIVATE /W4 /permissive-)
else()
    target_compile_options(underworld_sim PRIVATE -Wall -Wextra -Wpedantic)
endif()
option(UW_PYTHON "Build pybind11 interface" ON)
if(UW_PYTHON)
    set(PYBIND11_FINDPYTHON ON)
    find_package(Python 3.11 REQUIRED COMPONENTS Interpreter Development.Module)
    execute_process(COMMAND "${Python_EXECUTABLE}" -m pybind11 --cmakedir
        OUTPUT_VARIABLE pybind11_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    find_package(pybind11 3.0.4 EXACT CONFIG REQUIRED)
    pybind11_add_module(_underworld NO_EXTRAS core/bindings.cpp)
    target_link_libraries(_underworld PRIVATE underworld_sim)
endif()

add_executable(underworld_native_tests core/test_world.cpp)
target_link_libraries(underworld_native_tests PRIVATE underworld_sim)

add_executable(underworld_core core/main.cpp)
set_target_properties(underworld_core PROPERTIES
    CXX_STANDARD 20 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
if(MSVC)
    target_compile_options(underworld_core PRIVATE /W4 /permissive-)
else()
    target_compile_options(underworld_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME native_world COMMAND underworld_native_tests)
add_executable(underworld_thermal_tests core/test_thermal.cpp)
target_link_libraries(underworld_thermal_tests PRIVATE underworld_sim)
add_test(NAME native_thermal COMMAND underworld_thermal_tests)
add_test(NAME core_scene COMMAND underworld_core)
add_test(NAME core_reject_population COMMAND underworld_core --population 0)
set_tests_properties(core_reject_population PROPERTIES WILL_FAIL TRUE)
